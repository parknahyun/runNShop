**PaymentController.kt**
```kotlin
package com.runnshop.order.payment

import com.runnshop.order.payment.dto.PaymentRequest
import com.runnshop.order.payment.dto.PaymentResponse
import org.springframework.web.bind.annotation.*

@RestController
@RequestMapping("/api/v1/orders")
class PaymentController(
    private val paymentService: PaymentService
) {
    @PostMapping("/{orderId}/payment")
    fun pay(
        @PathVariable orderId: String,
        @RequestBody request: PaymentRequest
    ): PaymentResponse {
        return paymentService.pay(orderId, request)
    }
}
```

**PaymentService.kt**
```kotlin
package com.runnshop.order.payment

import com.runnshop.common.mock.MockLoader
import com.runnshop.order.payment.dto.PaymentRequest
import com.runnshop.order.payment.dto.PaymentResponse
import org.springframework.stereotype.Service

@Service
class PaymentService(
    private val mockLoader: MockLoader
) {
    // 프로토타입 단계: 실 PG·재고 연동 없이 Mock 성공 응답만 반환 (steering: coding-standards.md)
    fun pay(orderId: String, request: PaymentRequest): PaymentResponse {
        return mockLoader.load(
            "order__POST_orders_{orderId}_payment.json",
            PaymentResponse::class.java
        )
    }
}
```
